
// -----------------------------------------------------------
// 1. Create Constraints and Indexes
// -----------------------------------------------------------

CREATE CONSTRAINT category_id_unique IF NOT EXISTS
FOR (c:Category) REQUIRE c.category_id IS UNIQUE;

CREATE CONSTRAINT supplier_id_unique IF NOT EXISTS
FOR (s:Supplier) REQUIRE s.supplier_id IS UNIQUE;

CREATE CONSTRAINT store_id_unique IF NOT EXISTS
FOR (s:Store) REQUIRE s.store_id IS UNIQUE;

CREATE CONSTRAINT product_id_unique IF NOT EXISTS
FOR (p:Product) REQUIRE p.product_id IS UNIQUE;

CREATE CONSTRAINT customer_id_unique IF NOT EXISTS
FOR (c:Customer) REQUIRE c.customer_id IS UNIQUE;

CREATE CONSTRAINT staff_id_unique IF NOT EXISTS
FOR (s:Staff) REQUIRE s.staff_id IS UNIQUE;

CREATE CONSTRAINT order_id_unique IF NOT EXISTS
FOR (o:Order) REQUIRE o.order_id IS UNIQUE;

CREATE CONSTRAINT payment_id_unique IF NOT EXISTS
FOR (p:Payment) REQUIRE p.payment_id IS UNIQUE;


// -----------------------------------------------------------
// 2. Create Nodes

// Category
LOAD CSV WITH HEADERS FROM 'file:///Category.csv' AS row

SET c += row;

// -----------------------------------------------------------
// 3. Create Relationships
// -----------------------------------------------------------

// Product -> Category
LOAD CSV WITH HEADERS FROM 'file:///Product.csv' AS row
WITH row WHERE row.product_id IS NOT NULL AND row.category_id IS NOT NULL
MATCH (p:Product {product_id: row.product_id})
MATCH (c:Category {category_id: row.category_id})
MERGE (p)-[:BELONGS_TO]->(c);

// Product -> Supplier
LOAD CSV WITH HEADERS FROM 'file:///Product.csv' AS row
WITH row WHERE row.product_id IS NOT NULL AND row.supplier_id IS NOT NULL
MATCH (p:Product {product_id: row.product_id})
MATCH (s:Supplier {supplier_id: row.supplier_id})
MERGE (p)-[:SUPPLIED_BY]->(s);

// Staff -> Store
LOAD CSV WITH HEADERS FROM 'file:///Staff.csv' AS row
WITH row WHERE row.staff_id IS NOT NULL AND row.store_id IS NOT NULL
MATCH (s:Staff {staff_id: row.staff_id})
MATCH (st:Store {store_id: row.store_id})
MERGE (s)-[:WORKS_AT]->(st);

// Manager -> Store
LOAD CSV WITH HEADERS FROM 'file:///Manager.csv' AS row
WITH row WHERE row.staff_id IS NOT NULL
MATCH (m:Manager {staff_id: row.staff_id})
OPTIONAL MATCH (s:Staff {staff_id: row.staff_id})-[:WORKS_AT]->(st:Store)
WITH m, st
FOREACH (_ IN CASE WHEN st IS NULL THEN [] ELSE [1] END |
  MERGE (m)-[:MANAGES]->(st)
);

// Customer -> Order, Order -> Staff, Order -> Store
LOAD CSV WITH HEADERS FROM 'file:///Order.csv' AS row
WITH row WHERE row.order_id IS NOT NULL
MATCH (o:Order {order_id: row.order_id})
OPTIONAL MATCH (c:Customer {customer_id: row.customer_id})
WITH row, o, c
FOREACH (_ IN CASE WHEN c IS NULL THEN [] ELSE [1] END |
  MERGE (c)-[:PLACED]->(o)
)
WITH row, o
OPTIONAL MATCH (s:Staff {staff_id: row.staff_id})
WITH row, o, s
FOREACH (_ IN CASE WHEN s IS NULL THEN [] ELSE [1] END |
  MERGE (s)-[:PROCESSED]->(o)
)
WITH row, o
OPTIONAL MATCH (st:Store {store_id: row.store_id})
WITH o, st
FOREACH (_ IN CASE WHEN st IS NULL THEN [] ELSE [1] END |
  MERGE (o)-[:AT_STORE]->(st)
);

// Order -> Product
LOAD CSV WITH HEADERS FROM 'file:///OrderItem.csv' AS row
WITH row WHERE row.order_id IS NOT NULL AND row.product_id IS NOT NULL
MATCH (o:Order {order_id: row.order_id})
MATCH (p:Product {product_id: row.product_id})
MERGE (o)-[r:CONTAINS]->(p)
SET r += row;

// Payment -> Order
LOAD CSV WITH HEADERS FROM 'file:///Payment.csv' AS row
WITH row WHERE row.payment_id IS NOT NULL AND row.order_id IS NOT NULL
MATCH (pay:Payment {payment_id: row.payment_id})
MATCH (o:Order {order_id: row.order_id})
MERGE (pay)-[:PAID_FOR]->(o);

// Store -> Product (Inventory)
LOAD CSV WITH HEADERS FROM 'file:///Inventory.csv' AS row
WITH row WHERE coalesce(row.store_id,'') <> '' AND coalesce(row.product_id,'') <> ''
MATCH (st:Store {store_id: row.store_id})
MATCH (p:Product {product_id: row.product_id})
MERGE (st)-[r:STOCKS]->(p)
SET r += row;
